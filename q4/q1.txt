mysql> CREATE PROCEDURE Update_Sailor_Ratings()
    -> BEGIN
    ->   -- If rating < 5, increase by 2
    ->   UPDATE sailors
    ->   SET RATING = RATING + 2
    ->   WHERE RATING < 5
    ->   ;
    ->
    ->   -- If rating > 5 but < 10, increase by 1
    ->   UPDATE sailors
    ->   SET RATING = RATING + 1
    ->   WHERE RATING > 5
    ->     AND RATING < 10
    ->   ;
    ->
    ->   -- Ratings of 5 or 10 remain unchanged
    -> END$$

mysql> -- 4) Trigger: Cascade‐delete a customer’s related memberships & issues
mysql> -- ─────────────────────────────────────────────────────────────────────────────
mysql>
mysql> DELIMITER $$
mysql> CREATE TRIGGER trg_before_customer_delete
    -> BEFORE DELETE ON customer
    -> FOR EACH ROW
    -> BEGIN
    ->   -- delete all issue records for this customer's memberships
    ->   DELETE FROM iss_rec
    ->    WHERE mem_no IN (
    ->      SELECT mem_no
    ->      FROM membership
    ->      WHERE cust_no = OLD.cust_no
    ->    );
    ->   -- delete all membership records for this customer
    ->   DELETE FROM membership
    ->    WHERE cust_no = OLD.cust_no;
    -> END$$
